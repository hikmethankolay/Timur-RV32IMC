# forwarding_unit vectors
# Format: rs1 rs2 ex_rd ex_rw wb_rd wb_rw expA expB
#   rs1/rs2/ex_rd/wb_rd : hex  (5-bit, 2 digits)
#   ex_rw/wb_rw         : bin  (1-bit)
#   expA/expB           : bin  (2-bit)
#
# Encoding for forwardA/forwardB:
#   00 = regfile (no forward), 01 = MEM/WB, 10 = EX/MEM, 11 = unused

# T1 : idle - no producers writing
05 06 00 0 00 0 00 00

# T2 : producers exist but neither matches rs1/rs2
05 06 0a 1 0b 1 00 00

# T3 : EX/MEM hits rs1 only
05 06 05 1 00 0 10 00

# T4 : EX/MEM hits rs2 only
05 06 06 1 00 0 00 10

# T5 : EX/MEM hits both rs1 and rs2 (same rd)
05 05 05 1 00 0 10 10

# T6 : MEM/WB hits rs1, EX/MEM no match
05 06 0a 1 05 1 01 00

# T7 : MEM/WB hits rs2, EX/MEM no match
05 06 0a 1 06 1 00 01

# T8 : MEM/WB hits both rs1 and rs2
05 05 0a 1 05 1 01 01

# T9 : Priority - both producers write same reg, EX/MEM must win
05 06 05 1 05 1 10 00

# T10 : EX/MEM rd=x0 must NOT forward (all addrs zero, regwrite=1)
00 00 00 1 00 0 00 00

# T11 : MEM/WB rd=x0 must NOT forward
00 00 00 0 00 1 00 00

# T12 : EX/MEM regwrite=0 disables forwarding
05 06 05 0 00 0 00 00

# T13 : MEM/WB regwrite=0 disables forwarding (EX/MEM no match)
05 06 0a 1 05 0 00 00

# T14 : Mixed - rs1 from EX/MEM, rs2 from MEM/WB
05 06 05 1 06 1 10 01

# T15 : Mixed - rs1 from MEM/WB, rs2 from EX/MEM
05 06 06 1 05 1 01 10

# T16 : EX/MEM rd=x0 rejected, MEM/WB takes over for rs1
05 06 00 1 05 1 01 00

# T17 : EX/MEM rd=x1f (max), exercises full 5-bit width
1f 1e 1f 1 00 0 10 00

# T18 : Both rs read x0, x0 producer ignored on both sides
00 00 0a 1 0b 1 00 00
