# hazard_detection_unit vectors
# Format: memread id_ex_rd if_id_rs1 if_id_rs2 hready div_busy exp_stall
#   memread / hready / div_busy / exp_stall : bin (1-bit)
#   id_ex_rd / if_id_rs1 / if_id_rs2        : hex (5-bit, 2 digits)
#
# Stall sources:
#   load_use = memread && (rd != 0) && ((rd == rs1) || (rd == rs2))
#   bus_wait = !hready
#   div_busy = div_busy
#   stall    = load_use | bus_wait | div_busy

# === Idle ===
# T1 : nothing happening - no stall
0 00 05 06 1 0 0

# === Load-use hazard cases ===
# T2 : load to rd=5, consumer reads rs1=5  -> stall
1 05 05 06 1 0 1

# T3 : load to rd=6, consumer reads rs2=6  -> stall
1 06 05 06 1 0 1

# T4 : load to rd=5, consumer reads rs1=5 and rs2=5  -> stall
1 05 05 05 1 0 1

# T5 : load to rd=7, consumer reads rs1=5 rs2=6 (no match) -> no stall
1 07 05 06 1 0 0

# T6 : load to rd=x0, consumer reads rs1=x0 rs2=x0 -> NO stall (rd=0 exempt)
1 00 00 00 1 0 0

# T7 : load to rd=x0, consumer reads rs1=5 rs2=6 -> no stall
1 00 05 06 1 0 0

# T8 : non-load (memread=0), rd matches rs1 -> no stall
0 05 05 06 1 0 0

# T9 : non-load, rd matches both rs1 and rs2 -> no stall
0 05 05 05 1 0 0

# T10 : load, rd=x1f (max 5-bit), rs1=x1f -> stall
1 1f 1f 00 1 0 1

# T11 : load, rd=x1f, rs2=x1f -> stall
1 1f 00 1f 1 0 1

# === Bus wait alone ===
# T12 : hready=0, no other hazard -> stall
0 00 00 00 0 0 1

# T13 : hready=0, idle pipeline (load present but no match) -> stall
1 07 05 06 0 0 1

# === Divider busy alone ===
# T14 : div_busy=1, no other hazard -> stall
0 00 00 00 1 1 1

# T15 : div_busy=1, idle (load with no match) -> stall
1 07 05 06 1 1 1

# === Combined stall sources ===
# T16 : load_use AND bus_wait -> stall (still 1)
1 05 05 06 0 0 1

# T17 : load_use AND div_busy -> stall
1 05 05 06 1 1 1

# T18 : bus_wait AND div_busy -> stall
0 00 00 00 0 1 1

# T19 : all three sources active -> stall
1 05 05 06 0 1 1

# === Edge cases ===
# T20 : rd matches rs1 but memread=0 AND div_busy=1 -> stall (from div only)
0 05 05 06 1 1 1

# T21 : rd=x0 + bus_wait -> stall (from bus only)
1 00 00 00 0 0 1

# T22 : load, rd=5, rs1=4 rs2=6 (neither matches) -> no stall
1 05 04 06 1 0 0

# T23 : load, rd=x0, no other source -> no stall (x0 exempt)
1 00 05 00 1 0 0
