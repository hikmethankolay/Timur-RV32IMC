// format: op(dec) a(hex) b(hex) c(hex)
//   op 0  read       a = PADDR, b = expected PRDATA, c = compare mask
//   op 1  write      a = PADDR, b = PWDATA
//   op 2  loopback   a = 1: uart_rx follows uart_tx; 0: uart_rx driven by the testbench (idle high)
//   op 3  wait       a = cycles
//   op 4  TX frame   a = byte: the frame starts right after the preceding write; start bit, 8 data bits
//                    LSB first and stop bit are checked at their centres, every bit lasts 434 cycles and
//                    tx_busy is high for all 4340 cycles of the frame and low afterwards
//   op 5  RX frame   a = byte sent on uart_rx by the testbench with a valid stop bit
//   op 6  glitch     a = cycles uart_rx is held low (shorter than half a bit)
//   op 7  irq        a = expected irq
//   op 8  TX idle    a = cycles uart_tx must stay high (no frame)
//   op 9  RX frame with a bad stop bit (framing error), a = byte
// Registers: 0x00 DATA, 0x04 STATUS (bit 0 tx_busy, bit 1 rx_valid, bit 2 rx_overrun), 0x08 CTRL
