// rom_ahb test vectors
// The testbench first checks the ROM image against its own load of rom.hex, then fills
// the array with the pattern word(i) = {2'b01, i[13:0], 2'b10, i[13:0]} for these vectors.
// format: fetch_addr HSEL HTRANS HWRITE HREADY HADDR HWDATA check exp_fetch_instr exp_HRDATA wait_type
//   fetch_addr, HADDR, HWDATA, expected values: hex; HSEL, HWRITE, HREADY: bin; HTRANS: 2-bit bin
//   check: 2-bit bin, bit 0 = compare fetch_instr, bit 1 = compare HRDATA
//   HREADYOUT = 1 and HRESP = 0 are checked on every vector
// wait_type 0: check 3 ns after applying inputs, no clock edge; 1: next rising edge plus 1 ns
// fetch port: the word at address A appears after the edge that registers A
00000000 0 00 0 1 00000000 00000000 01 40008000 00000000 1
00000004 0 00 0 1 00000000 00000000 01 40018001 00000000 1
// same address presented again: same word (PC hold re-read)
00000004 0 00 0 1 00000000 00000000 01 40018001 00000000 1
// last word; only bits [15:2] index the array
0000FFFC 0 00 0 1 00000000 00000000 01 7FFFBFFF 00000000 1
00010008 0 00 0 1 00000000 00000000 01 40028002 00000000 1
// new address, no edge yet: output unchanged
00000010 0 00 0 1 00000000 00000000 01 40028002 00000000 0
00000010 0 00 0 1 00000000 00000000 01 40048004 00000000 1
// AHB port: NONSEQ read accepted with HSEL, HTRANS[1], HREADY
00000010 1 10 0 1 00000020 00000000 11 40048004 40088008 1
// IDLE transfer: not accepted, HRDATA holds
00000010 1 00 0 1 00000040 00000000 10 00000000 40088008 1
// HSEL = 0: not accepted
00000010 0 10 0 1 00000044 00000000 10 00000000 40088008 1
// HREADY = 0: not accepted
00000010 1 10 0 0 00000048 00000000 10 00000000 40088008 1
// SEQ transfer: accepted
00000010 1 11 0 1 0000004C 00000000 10 00000000 40138013 1
// write address phase: writes are ignored
00000010 1 10 1 1 00000050 00000000 00 00000000 00000000 1
// write data phase; fetch port still sees the original word
00000050 1 00 0 1 00000050 DEADBEEF 01 40148014 00000000 1
// read back through the AHB port: unchanged
00000050 1 10 0 1 00000050 00000000 10 00000000 40148014 1
// both ports on different words in the same cycle
00000100 1 10 0 1 00000200 00000000 11 40408040 40808080 1
00000104 1 10 0 1 00000204 00000000 11 40418041 40818081 1
// AHB port also indexes only bits [15:2]
00000108 1 10 0 1 00020008 00000000 11 40428042 40028002 1
